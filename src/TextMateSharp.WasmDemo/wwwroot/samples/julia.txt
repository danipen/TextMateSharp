module Demo

struct Person
    name::String
    age::Int
end

isadult(p::Person) = p.age >= 18

function greet(people::Vector{Person})
    for p in filter(isadult, people)
        println("Hello, $(p.name)!")
    end
end

greet([Person("Ada", 36), Person("Linus", 12)])
